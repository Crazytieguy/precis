  7→A simple OOP library for Lua. It has inheritance, metamethods (operators), class variables and weak mixin support.
  9→Quick Look
  10→==========
  11→
  12→```lua
  13→local class = require 'middleclass'
  14→
  15→local Fruit = class('Fruit') -- 'Fruit' is the class' name
  16→
  17→function Fruit:initialize(sweetness)
  18→  self.sweetness = sweetness
  19→end
  20→
  21→Fruit.static.sweetness_threshold = 5 -- class variable (also admits methods)
  22→
  23→function Fruit:isSweet()
  24→  return self.sweetness > Fruit.sweetness_threshold
  25→end
  26→
  27→local Lemon = class('Lemon', Fruit) -- subclassing
  28→
  29→function Lemon:initialize()
  30→  Fruit.initialize(self, 1) -- invoking the superclass' initializer
  31→end
  32→
  33→local lemon = Lemon:new()
  34→
  35→print(lemon:isSweet()) -- false
  36→```

  57→local function _propagateInstanceMethod(aClass, name, f)
  58→  f = name == "__index" and _createIndexWrapper(aClass, f) or f
  59→  aClass.__instanceDict[name] = f
  60→
  61→  for subclass in pairs(aClass.subclasses) do
  62→    if rawget(subclass.__declaredMethods, name) == nil then
  63→      _propagateInstanceMethod(subclass, name, f)
  64→    end
  65→  end

  109→local function _includeMixin(aClass, mixin)
  110→  assert(type(mixin) == 'table', "mixin must be a table")
  111→
  112→  for name,method in pairs(mixin) do
  113→    if name ~= "included" and name ~= "static" then aClass[name] = method end
  114→  end
  115→
  116→  for name,method in pairs(mixin.static or {}) do
  117→    aClass.static[name] = method
  118→  end
  119→
  120→  if type(mixin.included)=="function" then mixin:included(aClass) end
  121→  return aClass

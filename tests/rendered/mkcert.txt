  30→const shortUsage = `Usage of mkcert:
  31→
  32→	$ mkcert -install
  33→	Install the local CA in the system trust store.
  34→
  35→	$ mkcert example.org
  36→	Generate "example.org.pem" and "example.org-key.pem".
  37→
  38→	$ mkcert example.com myapp.dev localhost 127.0.0.1 ::1
  39→	Generate "example.com+4.pem" and "example.com+4-key.pem".
  40→
  41→	$ mkcert "*.example.it"
  42→	Generate "_wildcard.example.it.pem" and "_wildcard.example.it-key.pem".
  43→
  44→	$ mkcert -uninstall
  45→	Uninstall the local CA (but do not delete it).
  46→
  47→`

  50→
  51→	-cert-file FILE, -key-file FILE, -p12-file FILE
  52→	    Customize the output paths.
  53→
  54→	-client
  55→	    Generate a certificate for client authentication.
  56→
  57→	-ecdsa
  58→	    Generate a certificate with an ECDSA key.
  59→
  60→	-pkcs12
  61→	    Generate a ".p12" PKCS #12 file, also know as a ".pfx" file,
  62→	    containing certificate and key for legacy applications.
  63→
  64→	-csr CSR
  65→	    Generate a certificate based on the supplied CSR. Conflicts with
  66→	    all other flags and arguments except -install and -cert-file.
  67→
  68→	-CAROOT
  69→	    Print the CA certificate and key storage location.
  70→
  71→	$CAROOT (environment variable)
  72→	    Set the CA certificate and key storage location. (This allows
  73→	    maintaining multiple local CAs in parallel.)
  74→
  75→	$TRUST_STORES (environment variable)
  76→	    A comma-separated list of trust stores to install the local

    27→fn run() -> Result<()> {
    28→    let Some(args) = CommandArgs::parse()? else {
    29→        return Ok(());
    30→    };
    31→
    32→    let mut cfg = if args.ignore_config {
    33→        Config::default()
    34→    } else {
    35→        Config::load(args.config.clone())?
    36→    };
    37→
    38→    args.update_config(&mut cfg);
    39→    cfg.parse().context("parse config")?;
    40→
    41→    let cfg = Rc::new(cfg);
    42→
    43→    if args.show_config {
    44→        return cfg.show();
    45→    }
    46→
    47→    if let Some(ref debug_file) = args.debug {
    48→        debug::set_file(debug_file.clone());
    49→    }
    51→    // The user can specify the content type manually, or we can determine it based on the
    52→    // file extension.
    53→    // If the content type cannot be determined here, use `ContentType::Any` as default, let
    54→    // parser to detect it.
    55→    let content_type = args.get_content_type();
    56→
    57→    let max_data_size = args.max_data_size.unwrap_or(cfg.data.max_data_size) * 1024 * 1024;
    58→    let mut fw = None;
    59→    let data = match args.path.as_ref() {
    60→        Some(path) => {
    61→            let path = PathBuf::from(path);
    62→            if args.live_reload {
    63→                fw = Some(FileWatcher::new(
    64→                    path.clone(),
    65→                    cfg.clone(),
    66→                    content_type,
    67→                    max_data_size,
    68→                ));
    69→            }
    70→            fs::read(path).context("read file")?
    71→        }
    72→        None => {
    73→            let mut data = Vec::new();
    74→            io::stdin().read_to_end(&mut data).context("read stdin")?;
    75→            data
    76→        }
    77→    };
    79→    if let Some(target_type) = args.to {
    80→        if matches!(target_type, ContentType::Any) {
    81→            bail!("target content type cannot be 'any' when using '--to' option");
    82→        }
    83→        let parser = content_type.new_parser();
    84→        let target_parser = target_type.new_parser();
    85→
    86→        let value = parser.parse_root(Some(target_parser.as_ref()), &data)?;
    87→        let tokens = target_parser.syntax_highlight("", &value);
    88→        let text = SyntaxToken::pure_text(&tokens);
    89→
    90→        print!("{text}");
    91→        return Ok(());
    92→    }
    93→
    94→    if data.len() > max_data_size {
    95→        bail!("the data size is too large, we limit the maximum size to {} to ensure TUI performance, you should try to reduce the read size. HINT: You can use command line arg `--max-data-size` or config option `data.max_data_size` to modify this limitation", humansize::format_size(max_data_size, humansize::BINARY));
    96→    }
    98→    let tree = Tree::parse(cfg.clone(), &data, content_type).context("parse data")?;
    99→    let extension = tree.parser.extension();
    100→
    101→    let mut app = App::new(cfg.clone(), tree, fw);
    102→
    103→    if !cfg.header.disable {
    104→        let header_ctx = HeaderContext::new(args.path, extension, data.len());
    105→        app.set_header(&header_ctx);
    106→    }
    107→
    108→    ui::start(app)
    109→}
    111→fn main() {
    112→    if let Err(err) = run() {
    113→        eprintln!("Error: {err:#}");
    114→        process::exit(1);
    115→    }
    116→}
